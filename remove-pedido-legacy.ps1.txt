Param( [switch]$RunBuild )
$projectRoot = Split-Path -Parent $MyInvocation.MyCommand.Definition Set-Location $projectRoot
$timestamp = (Get-Date).ToString("yyyyMMdd-HHmmss") $backupDir = Join-Path $projectRoot "backups" New-Item -ItemType Directory -Path $backupDir -Force | Out-Null $zipPath = Join-Path $backupDir "backup-pedido-removal-$timestamp.zip"
$targets = @( "src\main\resources\templates\nuevo-venta-fixed.html", "src\main\resources\templates\nuevo-pedido.html", "src\main\resources\templates\confirmacion-pedido.html", "src\main\resources\templates\historial-pedidos.html", "build\resources\main\templates\nuevo-venta-fixed.html", "build\resources\main\templates\nuevo-pedido.html", "build\resources\main\templates\confirmacion-pedido.html", "build\resources\main\templates\confirmacion-venta.html", "src\main\java\edu\pe\utp\marcodesarrolloweb\ferrovoz\model\Pedido.java", "src\main\java\edu\pe\utp\marcodesarrolloweb\ferrovoz\dto\PedidoDTO.java", "src\main\java\edu\pe\utp\marcodesarrolloweb\ferrovoz\controller\PedidoController.java", "src\main\java\edu\pe\utp\marcodesarrolloweb\ferrovoz\service\PedidoService.java", "src\main\java\edu\pe\utp\marcodesarrolloweb\ferrovoz\repository\PedidoRepository.java" )
$existing = $targets | Where-Object { Test-Path $_ } if ($existing.Count -eq 0) { Write-Host "No se encontraron archivos objetivo. Nada que hacer." -ForegroundColor Yellow exit 0 }
Write-Host "Archivos que se respaldarán y eliminarán:" -ForegroundColor Cyan $existing | ForEach-Object { Write-Host " - $_" }
try { Compress-Archive -Path $existing -DestinationPath $zipPath -Force Write-Host "Backup creado: $zipPath" -ForegroundColor Green } catch { Write-Error "Error creando backup: $_" exit 1 }
foreach ($f in $existing) { try { Remove-Item -Force -LiteralPath $f -Verbose } catch { Write-Warning "No se pudo borrar $f : $_" } }
if ($RunBuild) { if (Test-Path ".\gradlew.bat") { Write-Host "Ejecutando wrapper gradle clean..." -ForegroundColor Cyan & .\gradlew.bat clean } else { Write-Host "Ejecutando gradle clean (asegúrate de tener gradle instalado)..." -ForegroundColor Cyan gradle clean } }
Write-Host "Operación completada. Revisa $zipPath para restaurar si hace falta." -ForegroundColor Green
